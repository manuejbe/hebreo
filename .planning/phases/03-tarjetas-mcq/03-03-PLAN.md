**Step 1: getDistractors function** (add before renderOpcionMultiple):

```js
/**
 * Returns 3 distractor words for the MCQ question.
 * Priority: same kind + same cat. For countries: same group (region).
 * Fallback: same kind, any cat. Final fallback: any word, different text.
 * Never duplicates the correct word's visible text. (MCQ-01, MCQ-02)
 * @param {Object} correct — the correct word
 * @param {'es-he'|'he-es'} dir — resolved direction
 * @returns {Object[]} array of 3 distractor word objects
 */
function getDistractors(correct, dir) {
  const correctText = dir === 'es-he' ? correct.msg : correct.es;
  const usedTexts = new Set([correctText]);

  function getText(w) { return dir === 'es-he' ? w.msg : w.es; }
  function isValid(w) {
    if (w.id === correct.id) return false;
    const t = getText(w);
    if (!t || usedTexts.has(t)) return false;
    return true;
  }
  function pick(pool) {
    const result = [];
    // Shuffle pool
    const arr = pool.slice();
    for (let i = arr.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    for (const w of arr) {
      if (isValid(w) && result.length < 3 - result.length + result.length) {
        usedTexts.add(getText(w));
        result.push(w);
        if (result.length === 3) break;
      }
    }
    return result;
  }

  // Simpler pick:
  function pickN(pool, n) {
    const result = [];
    const arr = pool.slice();
    for (let i = arr.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    for (const w of arr) {
      if (result.length >= n) break;
      if (!isValid(w)) continue;
      usedTexts.add(getText(w));
      result.push(w);
    }
    return result;
  }

  const vocab = App.state.vocab;
  let distractors = [];

  if (correct.kind === 'country' && correct.group) {
    // Same region first (MCQ-02)
    const sameRegion = vocab.filter(w => w.kind === 'country' && w.group === correct.group);
    distractors = pickN(sameRegion, 3);
  }

  if (distractors.length < 3) {
    // Fill from same kind + same cat
    const sameCat = vocab.filter(w => w.kind === correct.kind && w.cat === correct.cat);
    distractors = distractors.concat(pickN(sameCat, 3 - distractors.length));
  }

  if (distractors.length < 3) {
    // Fallback: same kind, any cat
    const sameKind = vocab.filter(w => w.kind === correct.kind);
    distractors = distractors.concat(pickN(sameKind, 3 - distractors.length));
  }

  if (distractors.length < 3) {
    // Final fallback: any word
    distractors = distractors.concat(pickN(vocab, 3 - distractors.length));
  }

  return distractors.slice(0, 3);
}
```

```js
App.ui.renderOpcionMultiple = function(el) {
  if (!App.session || !App.session.words || App.session.words.length === 0) {
    location.hash = '#/sesion'; return;
  }
  if (App.session.index >= App.session.words.length) {
    App.ui.renderResultado(el); return;
  }

  const total   = App.session.words.length;
  const idx     = App.session.index;
  const word    = App.session.words[idx];
  const pct     = (idx / total) * 100;

  // Resolve direction
  const dir = App.session.direction === 'mezclada'
    ? (Math.random() < 0.5 ? 'es-he' : 'he-es')
    : App.session.direction;
  App.session._cardDir = dir;

  // Question text and option text extractor
  const questionText = dir === 'es-he' ? word.es : (word.msg || word.es);
  const questionIsHe = dir === 'he-es';
  function getOptionText(w) { return dir === 'es-he' ? (w.msg || '?') : w.es; }
  const correctText = getOptionText(word);

  // Build 4 options: 1 correct + 3 distractors, shuffled
  const distractors = getDistractors(word, dir);
  const options = [{ word, correct: true }, ...distractors.map(w => ({ word: w, correct: false }))];
  // Shuffle
  for (let i = options.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [options[i], options[j]] = [options[j], options[i]];
  }

  let answered = false;

  el.innerHTML = '';

  // Session progress bar (DESIGN.md §3.7)
  const progBar = document.createElement('div');
  progBar.className = 'session-progress';
  const progFill = document.createElement('div');
  progFill.className = 'session-progress-fill';
  progFill.style.width = pct + '%';
  progBar.appendChild(progFill);
  el.appendChild(progBar);

  // Counter
  const counter = document.createElement('p');
  counter.style.cssText = 'text-align:center;font:600 var(--text-sm) var(--font-body);color:var(--text-muted);padding:var(--sp-6) var(--sp-4) var(--sp-3);';
  counter.textContent = (idx + 1) + ' / ' + total;
  el.appendChild(counter);

  // Question
  const qSection = document.createElement('section');
  qSection.style.cssText = 'padding:0 var(--sp-4);margin-bottom:var(--sp-5);';
  const qEl = document.createElement('p');
  qEl.style.cssText = 'font:600 1.5rem var(--font-display);color:var(--text-primary);text-align:center;line-height:1.3;';
  if (questionIsHe) {
    qEl.lang = 'he'; qEl.dir = 'rtl';
    qEl.style.fontFamily = 'var(--font-hebrew)';
    qEl.style.fontSize = 'calc(var(--he-scale)*1.5rem)';
  }
  qEl.textContent = questionText;  // textContent — XSS safe
  qSection.appendChild(qEl);
  el.appendChild(qSection);

  // Options grid 2x2 (DESIGN.md §3.2)
  const grid = document.createElement('div');
  grid.style.cssText = 'display:grid;grid-template-columns:1fr 1fr;gap:var(--sp-3);padding:0 var(--sp-4);margin-bottom:var(--sp-5);';
  // Single column on very narrow (<320px)
  if (!document.getElementById('mcq-grid-style')) {
    const s = document.createElement('style');
    s.id = 'mcq-grid-style';
    s.textContent = '@media (max-width:320px) { .mcq-grid { grid-template-columns:1fr !important; } }';
    document.head.appendChild(s);
  }
  grid.classList.add('mcq-grid');

  const optionEls = [];

  options.forEach((opt, i) => {
    const btn = document.createElement('button');
    btn.style.cssText = 'position:relative;background:var(--surface);border:1.5px solid var(--border);border-radius:var(--radius-md);min-height:64px;padding:var(--sp-4) var(--sp-3) var(--sp-4) var(--sp-8);text-align:left;cursor:pointer;transition:border-color var(--dur-fast),background var(--dur-fast);font:400 var(--text-sm) var(--font-body);color:var(--text-primary);';
    btn.setAttribute('aria-label', 'Opcion ' + (i + 1) + ': ' + getOptionText(opt.word));

    // Shortcut chip (DESIGN.md §3.2: chip 12x18px, blue-100/blue-700)
    const chip = document.createElement('span');
    chip.setAttribute('aria-hidden', 'true');
    chip.style.cssText = 'position:absolute;top:var(--sp-2);left:var(--sp-2);width:18px;height:18px;background:var(--blue-100);color:var(--blue-700);border-radius:3px;font:600 var(--text-xs) var(--font-body);display:flex;align-items:center;justify-content:center;font-size:0.65rem;';
    chip.textContent = i + 1;
    btn.appendChild(chip);

    // Option text
    const textEl = document.createElement('span');
    const optText = getOptionText(opt.word);
    if (dir === 'es-he') {
      // Showing Hebrew options
      textEl.lang = 'he'; textEl.dir = 'rtl';
      textEl.style.cssText = 'font-family:var(--font-hebrew);font-size:calc(var(--he-scale)*0.95rem);direction:rtl;display:block;text-align:right;';
    } else {
      textEl.style.cssText = 'font:400 var(--text-sm) var(--font-body);';
    }
    textEl.textContent = optText;  // textContent — XSS safe
    btn.appendChild(textEl);

    // Hover/focus (DESIGN.md §3.2)
    btn.addEventListener('mouseenter', () => { if (!answered) { btn.style.borderColor='var(--blue-600)'; btn.style.background='var(--blue-50)'; } });
    btn.addEventListener('mouseleave', () => { if (!answered) { btn.style.borderColor='var(--border)'; btn.style.background='var(--surface)'; } });

    btn.addEventListener('click', () => handleAnswer(i, opt.correct));
    optionEls.push(btn);
    grid.appendChild(btn);
  });

  el.appendChild(grid);

  // "Siguiente" button — hidden until answered
  const nextBtn = document.createElement('button');
  nextBtn.style.cssText = 'display:none;width:calc(100% - var(--sp-8));margin:0 var(--sp-4);padding:var(--sp-3) var(--sp-4);background:var(--blue-700);color:#fff;border-radius:var(--radius-md);font:700 var(--text-base) var(--font-body);border:none;cursor:pointer;transition:background var(--dur-fast);';
  nextBtn.textContent = 'Siguiente';
  nextBtn.setAttribute('aria-label', 'Siguiente pregunta');
  nextBtn.addEventListener('mouseenter', () => { nextBtn.style.background = 'var(--blue-800)'; });
  nextBtn.addEventListener('mouseleave', () => { nextBtn.style.background = 'var(--blue-700)'; });
  nextBtn.addEventListener('click', advance);
  el.appendChild(nextBtn);

  // --- ANSWER LOGIC ---

  function handleAnswer(chosenIdx, isCorrect) {
    if (answered) return;
    answered = true;

    App.ui.updateProgress(word.id, isCorrect);
    App.session.results.push({ wordId: word.id, correct: isCorrect });

    // Apply visual feedback (DESIGN.md §3.2 post-respuesta)
    options.forEach((opt, i) => {
      const btn = optionEls[i];
      btn.style.cursor = 'default';
      if (opt.correct) {
        // Correct option: green (MCQ-03)
        btn.style.background   = 'var(--green-bg)';
        btn.style.borderColor  = 'var(--green)';
        btn.style.color        = 'var(--green)';
      } else if (i === chosenIdx && !isCorrect) {
        // Chosen wrong option: red (MCQ-03)
        btn.style.background   = 'var(--red-bg)';
        btn.style.borderColor  = 'var(--red)';
        btn.style.color        = 'var(--red)';
      } else {
        // Rest: dim (MCQ-03: opacity .45)
        btn.style.opacity = '.45';
        btn.style.pointerEvents = 'none';
      }
    });

    // Flash/shake animation on question (DESIGN.md §3.6)
    if (isCorrect) {
      qSection.style.animation = 'flash-green 350ms ease-out';
    } else {
      qSection.style.animation = 'shake 300ms ease-out';
    }
    qSection.addEventListener('animationend', () => { qSection.style.animation = ''; }, { once: true });

    // Show "Siguiente" button
    nextBtn.style.display = 'block';
  }

  function advance() {
    App.session.index++;
    App.ui.renderOpcionMultiple(el);
  }

  // Keyboard shortcuts (MCQ-04: 1-4 choose option, Enter advances)
  function onKey(e) {
    if (e.key >= '1' && e.key <= '4') {
      const i = parseInt(e.key, 10) - 1;
      if (i < options.length) {
        e.preventDefault();
        if (!answered) handleAnswer(i, options[i].correct);
      }
    }
    if (e.key === 'Enter') {
      e.preventDefault();
      if (answered) advance();
    }
  }
  document.addEventListener('keydown', onKey);
  // Cleanup on navigation
  const observer = new MutationObserver(() => {
    if (!el.isConnected) { document.removeEventListener('keydown', onKey); observer.disconnect(); }
  });
  observer.observe(document.getElementById('app-main'), { childList: true });
};
```
