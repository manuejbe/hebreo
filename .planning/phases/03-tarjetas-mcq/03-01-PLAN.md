```js
App.session.start = function(mode) {
  const st = App.ui._sesionState;   // { cats, filter, direction, quantity }
  // Compute matching words (same logic already in renderSesion's getMatchingWords)
  const catOrder = [];
  const catSeen = new Set();
  for (const w of App.state.vocab) { if (!catSeen.has(w.cat)) { catOrder.push(w.cat); catSeen.add(w.cat); } }
  const selected = st.cats.length === 0 ? catOrder : st.cats;
  const now = Date.now();
  const sevenDaysAgo = now - 7 * 24 * 60 * 60 * 1000;
  let words = App.state.vocab.filter(w => selected.includes(w.cat));
  if (st.filter === 'nuevas') {
    words = words.filter(w => !App.state.progress[w.id] || App.state.progress[w.id].box === 0);
  } else if (st.filter === 'dificiles') {
    words = words.filter(w => { const p = App.state.progress[w.id]; if (!p) return true; return p.box <= 2 || (p.lastWrong && p.lastWrong >= sevenDaysAgo); });
  } else if (st.filter === 'hoy') {
    words = words.filter(w => { const p = App.state.progress[w.id]; return p && p.box > 0 && p.due <= now; });
  }
  // Shuffle
  for (let i = words.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [words[i], words[j]] = [words[j], words[i]];
  }
  const qty = st.quantity === 0 ? words.length : st.quantity;
  App.session = {
    mode,
    words: words.slice(0, qty),
    index: 0,
    direction: st.direction,
    results: []    // { wordId, correct: bool }
  };
};
```

**Leitner update function** — add `App.ui.updateProgress(wordId, correct)` in the same section:
```js
App.ui.updateProgress = function(wordId, correct) {
  const p = App.state.progress[wordId] || { box: 0, due: 0, seen: 0, correct: 0, wrong: 0, lastWrong: null, streak: 0 };
  p.seen++;
  if (correct) {
    p.correct++;
    p.streak = (p.streak || 0) + 1;
    p.box = Math.min(p.box + 1, 5);          // per LEIT-02: sube una caja
  } else {
    p.wrong++;
    p.streak = 0;
    p.lastWrong = Date.now();
    p.box = 1;                                 // per LEIT-02: vuelve a caja 1
  }
  // due timestamp placeholder (full intervals come in Phase 5)
  p.due = Date.now();
  App.state.progress[wordId] = p;
  Store.set('ivrit.v1.progress', App.state.progress);  // persist after every answer
};
```

```js
/**
 * Renders session result screen. Called by renderTarjetas and renderOpcionMultiple.
 * @param {HTMLElement} el — screen container
 */
App.ui.renderResultado = function(el) {
  const { results, words, mode } = App.session;
  const total   = results.length;
  const correct = results.filter(r => r.correct).length;
  const wrong   = results.filter(r => !r.correct);

  el.innerHTML = '';

  // Header
  const header = document.createElement('header');
  header.style.cssText = 'padding:var(--sp-8) var(--sp-4) var(--sp-6);text-align:center;';
  const pct = total > 0 ? Math.round((correct / total) * 100) : 0;
  const title = document.createElement('h1');
  title.style.cssText = 'font:600 3rem var(--font-display);color:var(--text-primary);margin-bottom:var(--sp-2);';
  title.textContent = pct >= 80 ? '¡Muy bien!' : pct >= 50 ? 'Buen progreso' : 'Sigue practicando';
  header.appendChild(title);

  const sub = document.createElement('p');
  sub.style.cssText = 'font:400 var(--text-base) var(--font-body);color:var(--text-muted);';
  sub.textContent = correct + ' de ' + total + ' correctas (' + pct + '%)';
  header.appendChild(sub);
  el.appendChild(header);

  // Wrong words list
  if (wrong.length > 0) {
    const section = document.createElement('section');
    section.style.cssText = 'padding:0 var(--sp-4);margin-bottom:var(--sp-6);';
    const h2 = document.createElement('h2');
    h2.style.cssText = 'font:600 1.125rem var(--font-display);color:var(--text-primary);margin-bottom:var(--sp-3);';
    h2.textContent = 'Palabras a repasar (' + wrong.length + ')';
    section.appendChild(h2);

    const list = document.createElement('ul');
    list.style.cssText = 'list-style:none;display:flex;flex-direction:column;gap:var(--sp-2);';
    for (const r of wrong) {
      const w = App.state.vocab.find(v => v.id === r.wordId);
      if (!w) continue;
      const li = document.createElement('li');
      li.style.cssText = 'display:flex;justify-content:space-between;align-items:center;padding:var(--sp-3) var(--sp-4);background:var(--surface);border:1px solid var(--border);border-radius:var(--radius-md);gap:var(--sp-3);';
      const esSpan = document.createElement('span');
      esSpan.style.cssText = 'font:600 var(--text-sm) var(--font-body);color:var(--text-primary);';
      esSpan.textContent = w.es;    // textContent — XSS safe
      const heSpan = document.createElement('span');
      heSpan.lang = 'he'; heSpan.dir = 'rtl';
      heSpan.style.cssText = 'font-family:var(--font-hebrew);font-size:calc(var(--he-scale)*0.95rem);color:var(--text-secondary);';
      heSpan.textContent = w.msg || '';  // textContent — XSS safe
      li.appendChild(esSpan);
      li.appendChild(heSpan);
      list.appendChild(li);
    }
    section.appendChild(list);
    el.appendChild(section);
  }

  // Action buttons
  const btnArea = document.createElement('div');
  btnArea.style.cssText = 'padding:0 var(--sp-4);display:flex;flex-direction:column;gap:var(--sp-3);';

  // "Repetir las que no sabia" — only if there are wrong words
  if (wrong.length > 0) {
    const retryBtn = document.createElement('button');
    retryBtn.style.cssText = 'display:block;width:100%;padding:var(--sp-4);background:var(--blue-700);color:#fff;border-radius:var(--radius-lg);font:700 var(--text-lg) var(--font-body);border:none;cursor:pointer;transition:background var(--dur-fast);';
    retryBtn.textContent = 'Repetir las que no sabia (' + wrong.length + ')';
    retryBtn.setAttribute('aria-label', 'Repetir ' + wrong.length + ' palabras incorrectas');
    retryBtn.addEventListener('mouseenter', () => { retryBtn.style.background = 'var(--blue-800)'; });
    retryBtn.addEventListener('mouseleave', () => { retryBtn.style.background = 'var(--blue-700)'; });
    retryBtn.addEventListener('click', () => {
      // Start a new session with only the wrong words
      App.session.words  = wrong.map(r => App.state.vocab.find(v => v.id === r.wordId)).filter(Boolean);
      App.session.index  = 0;
      App.session.results = [];
      // Navigate to the same mode
      location.hash = '#/' + (App.session.mode || 'tarjetas');
    });
    btnArea.appendChild(retryBtn);
  }

  const newBtn = document.createElement('button');
  newBtn.style.cssText = 'display:block;width:100%;padding:var(--sp-3) var(--sp-4);border:1.5px solid var(--border);border-radius:var(--radius-lg);font:600 var(--text-base) var(--font-body);background:transparent;color:var(--text-secondary);cursor:pointer;transition:background var(--dur-fast);';
  newBtn.textContent = 'Nueva sesion';
  newBtn.addEventListener('mouseenter', () => { newBtn.style.background = 'var(--surface-hover)'; });
  newBtn.addEventListener('mouseleave', () => { newBtn.style.background = 'transparent'; });
  newBtn.addEventListener('click', () => { location.hash = '#/sesion'; });
  btnArea.appendChild(newBtn);

  el.appendChild(btnArea);
};
```
