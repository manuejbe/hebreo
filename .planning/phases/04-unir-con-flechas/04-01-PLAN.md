  function shuffleArr(arr) {
    const a = arr.slice();
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];

  const resolvedPairs = {};  // wordId → { leftEl, rightEl, color }

  function redrawLines() {
    // Limpiar SVG y redibujar todas las líneas persistentes
    while (svg.firstChild) svg.removeChild(svg.firstChild);
    for (const wid of Object.keys(resolvedPairs)) {
      const { leftEl, rightEl, color } = resolvedPairs[wid];
      lines[wid] = drawLine(leftEl, rightEl, color);
    }
    // Si hay una línea temporal pendiente (pendingLine), también redibujar
    if (pendingLine && pendingLine.leftEl && pendingLine.rightEl) {
      pendingLine.line = drawLine(pendingLine.leftEl, pendingLine.rightEl, pendingLine.color);

  const onResize = () => redrawLines();

**Lógica tap-tap:**

```javascript
  let pendingLine = null;  // { leftEl, rightEl, line, color } — línea temporal de error

  function handleTap(clickedEl) {
    const wordId = clickedEl.dataset.wordId;
    const side   = clickedEl.dataset.side;

    // No permitir re-selección de elementos ya resueltos
    if (clickedEl.dataset.resolved === 'true') return;

    if (side === 'left') {
      // Deselect si es el mismo (D-02)
      if (selectedLeft && selectedLeft.el === clickedEl) {
        clickedEl.style.borderColor = 'var(--border)';
        clickedEl.style.background  = 'var(--surface)';
        selectedLeft = null;
        return;
      }
      // Cambiar selección en la misma columna (D-02)
      if (selectedLeft) {
        selectedLeft.el.style.borderColor = 'var(--border)';
        selectedLeft.el.style.background  = 'var(--surface)';
      }
      selectedLeft = { el: clickedEl, wordId };
      clickedEl.style.borderColor = 'var(--blue-700)';
      clickedEl.style.background  = 'var(--blue-50, #EFF6FF)';
    } else {
      // side === 'right'
      if (!selectedLeft) {
        // No hay elemento izquierdo seleccionado — deselect propio si aplica, o noop
        if (selectedRight && selectedRight.el === clickedEl) {
          clickedEl.style.borderColor = 'var(--border)';
          clickedEl.style.background  = 'var(--surface)';
          selectedRight = null;
        } else {
          if (selectedRight) { selectedRight.el.style.borderColor='var(--border)'; selectedRight.el.style.background='var(--surface)'; }
          selectedRight = { el: clickedEl, wordId };
          clickedEl.style.borderColor = 'var(--blue-700)';
          clickedEl.style.background  = 'var(--blue-50, #EFF6FF)';
        }
        return;
      }

      // Tenemos selectedLeft — evaluar par
      const isCorrect = selectedLeft.wordId === wordId;
      const color     = isCorrect ? 'var(--green)' : 'var(--red)';

      // Dibujar línea
      const lineEl = drawLine(selectedLeft.el, clickedEl, color);

      if (isCorrect) {
        // Correcto: atenuar ambos, marcar resueltos, registrar
        [selectedLeft.el, clickedEl].forEach(e => {
          e.style.opacity       = '0.45';
          e.style.borderColor   = 'var(--green)';
          e.style.background    = 'var(--green-bg)';
          e.dataset.resolved    = 'true';
          e.style.pointerEvents = 'none';
          e.style.animation     = 'flash-green 350ms ease-out';
          e.addEventListener('animationend', () => { e.style.animation=''; }, { once: true });
        });
        resolvedPairs[selectedLeft.wordId] = { leftEl: selectedLeft.el, rightEl: clickedEl, color: 'var(--green)' };
        lines[selectedLeft.wordId] = lineEl;

        // Leitner — solo ES↔HE (D-10)
        if (!App.session._matchVariant || App.session._matchVariant === 'es-he') {
          App.ui.updateProgress(selectedLeft.wordId, true);
        }
        App.session.results.push({ wordId: selectedLeft.wordId, correct: true });

        resolved++;
        selectedLeft  = null;
        selectedRight = null;

        // ¿Ronda terminada?
        if (resolved >= pairs.length) {
          showRoundSummary();
        }
      } else {
        // Incorrecto: shake + rojo + limpiar tras 400ms
        roundErrors++;
        [selectedLeft.el, clickedEl].forEach(e => {
          e.style.borderColor = 'var(--red)';
          e.style.background  = 'var(--red-bg)';
          e.style.animation   = 'shake 400ms ease-out';
        });

        // Leitner — solo ES↔HE (D-10/D-11)
        if (!App.session._matchVariant || App.session._matchVariant === 'es-he') {
          App.ui.updateProgress(selectedLeft.wordId, false);
        }
        App.session.results.push({ wordId: selectedLeft.wordId, correct: false });

        const capturedLeft  = selectedLeft;
        const capturedRight = { el: clickedEl, wordId };
        pendingLine = { leftEl: capturedLeft.el, rightEl: clickedEl, line: lineEl, color: 'var(--red)' };
        selectedLeft  = null;
        selectedRight = null;

        setTimeout(() => {
          [capturedLeft.el, capturedRight.el].forEach(e => {
            e.style.borderColor = 'var(--border)';
            e.style.background  = 'var(--surface)';
            e.style.animation   = '';
          });
          if (lineEl.parentNode) lineEl.parentNode.removeChild(lineEl);
          pendingLine = null;
        }, 450);  // 400ms shake + 50ms buffer
      }
    }
  }
```
