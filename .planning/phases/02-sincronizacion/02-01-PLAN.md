  **App.sync.fetchWithTimeout(url, ms)**
  ```javascript
  App.sync.fetchWithTimeout = async function(url, ms) {
    if (!navigator.onLine) throw Object.assign(new Error('sin conexión'), { code: 'offline' });
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), ms);
    try {
      const resp = await fetch(url, { signal: ctrl.signal });
      clearTimeout(timer);
      if (!resp.ok) throw Object.assign(new Error('URL no responde'), { code: 'timeout' });
      return await resp.json();
    } catch (err) {
      clearTimeout(timer);
      if (err.code) throw err;
      if (err.name === 'AbortError') throw Object.assign(new Error('La URL no respondió en 20 s'), { code: 'timeout' });
      throw Object.assign(new Error('sin conexión'), { code: 'offline' });
    }
  };
  ```

  **App.sync.validateRaw(data)**
  Check: `data && data.source === 'google-docs' && Array.isArray(data.categories) && data.categories.length > 0 && data.categories.every(c => typeof c.name === 'string' && Array.isArray(c.headers) && Array.isArray(c.rows))`. Returns boolean.

  **App.sync.diffVocab(oldWords, newWords)**
  Build a Map from each array keyed by id. Compare:
  - added: newWords where id not in oldMap
  - deleted: oldWords where id not in newMap
  - modified: ids in both where any of [es, msg, mpl, fsg, fpl, group] differs (strict equality)
  Returns `{ added, modified, deleted }`.

  **App.sync.applyDiff(diff, rawData)**
  - Push diff.added to App.state.vocab
  - For each diff.modified: find index in App.state.vocab by id, replace element
  - For each diff.deleted: find index in App.state.vocab by id, splice out (leave progress[id] intact — SYNC-04)
  - Store.set('ivrit.v1.vocabCache', { raw: rawData, updated: rawData.updated || new Date().toISOString() })

  **App.sync.run()**
  ```javascript
  App.sync.run = async function() {
    let raw;
    try {
      raw = await App.sync.fetchWithTimeout(APPS_SCRIPT_URL, 20000);
    } catch (err) {
      throw { code: err.code || 'offline', message: err.message };
    }
    if (!App.sync.validateRaw(raw)) {
      throw { code: 'invalid', message: 'El formato de la respuesta no es válido' };
    }
    let newWords;
    try {
      newWords = App.data.normalize(raw);
    } catch (_) {
      throw { code: 'invalid', message: 'Error al procesar los datos recibidos' };
    }
    const diff = App.sync.diffVocab(App.state.vocab, newWords);
    App.sync.applyDiff(diff, raw);
    return { diff, updatedAt: raw.updated || new Date().toISOString() };
  };
  ```
