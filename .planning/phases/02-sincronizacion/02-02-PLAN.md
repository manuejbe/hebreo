  Insert in the `=== 4. UTILIDADES DE UI ===` section (after showToast, before the next section). Add:

  ```javascript
  /**
   * Runs sync, manages button state, shows result panel.
   * Called from the Sincronizar button onclick in renderInicio.
   */
  App.ui.syncRun = async function(btnEl) {
    if (btnEl) {
      btnEl.disabled = true;
      btnEl.innerHTML = '<span style="display:inline-block;width:16px;height:16px;border:2px solid var(--blue-200);border-top-color:var(--blue-700);border-radius:50%;animation:spin 0.7s linear infinite;vertical-align:middle;margin-right:6px;"></span>Sincronizando…';
    }
    try {
      const { diff, updatedAt } = await App.sync.run();
      App.ui.renderSyncResult(diff, updatedAt);
    } catch (err) {
      const msgs = {
        offline: 'Sin conexión — revisa tu internet',
        timeout: 'URL no responde — intenta más tarde',
        invalid: 'Formato no válido — revisa la URL del Apps Script'
      };
      App.ui.showToast('⚠ ' + (msgs[err.code] || err.message || 'Error desconocido'), 5000);
    } finally {
      if (btnEl) {
        btnEl.disabled = false;
        btnEl.textContent = '🔄 Sincronizar';
      }
    }
  };

  /**
   * Shows a result panel (modal-style overlay) after a successful sync.
   * All remote word data rendered via textContent (XSS-safe).
   */
  App.ui.renderSyncResult = function(diff, updatedAt) {
    // Remove any existing result panel
    const existing = document.getElementById('sync-result-panel');
    if (existing) existing.remove();

    const { added, modified, deleted } = diff;
    const hasChanges = added.length + modified.length + deleted.length > 0;

    // Format updated date
    let dateStr = '';
    try { dateStr = new Date(updatedAt).toLocaleDateString('es', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' }); } catch { dateStr = updatedAt; }

    // Update the lastSync span in Inicio without full re-render
    const syncDateEl = document.getElementById('sync-last-date');
    if (syncDateEl) syncDateEl.textContent = dateStr;

    const overlay = document.createElement('div');
    overlay.id = 'sync-result-panel';
    overlay.style.cssText = 'position:fixed;inset:0;background:rgba(26,24,20,.45);backdrop-filter:blur(2px);z-index:300;display:flex;align-items:flex-end;justify-content:center;padding:var(--sp-4);animation:sync-overlay-in 250ms ease both;';

    const panel = document.createElement('div');
    panel.style.cssText = 'background:var(--surface);border-radius:var(--radius-lg);box-shadow:var(--shadow-lg);max-width:400px;width:100%;padding:var(--sp-6);animation:sync-panel-in 250ms var(--ease-out) both;max-height:80vh;overflow-y:auto;';

    // Summary line (all text via textContent nodes — XSS safe)
    const summary = document.createElement('p');
    summary.style.cssText = 'font:600 var(--text-base) var(--font-body);color:var(--text-primary);margin-bottom:var(--sp-4);';
    if (hasChanges) {
      summary.textContent = '✅ ' + added.length + ' nuevas · ' + modified.length + ' modificadas · ' + deleted.length + ' eliminadas';
    } else {
      summary.textContent = '✅ Sin cambios — el vocabulario está actualizado';
    }
    panel.appendChild(summary);

    // New words collapsible list (SYNC-03)
    if (added.length > 0) {
      const details = document.createElement('details');
      details.style.cssText = 'border:1px solid var(--border);border-radius:var(--radius-md);margin-bottom:var(--sp-4);overflow:hidden;';
      const summ = document.createElement('summary');
      summ.style.cssText = 'padding:var(--sp-3) var(--sp-4);cursor:pointer;font:600 var(--text-sm) var(--font-body);color:var(--text-primary);list-style:none;background:var(--surface);';
      summ.textContent = added.length + ' palabras nuevas ▸';
      details.appendChild(summ);
      const list = document.createElement('ul');
      list.style.cssText = 'margin:0;padding:var(--sp-2) var(--sp-4) var(--sp-3);list-style:none;font-size:var(--text-sm);display:flex;flex-direction:column;gap:4px;';
      for (const w of added) {
        const li = document.createElement('li');
        li.style.cssText = 'display:flex;justify-content:space-between;align-items:center;gap:var(--sp-3);';
        const esSpan = document.createElement('span');
        esSpan.style.cssText = 'color:var(--text-secondary);';
        esSpan.textContent = w.es;             // textContent — XSS safe
        const heSpan = document.createElement('span');
        heSpan.lang = 'he';
        heSpan.dir = 'rtl';
        heSpan.className = 'he';
        heSpan.style.cssText = 'font-size:calc(var(--he-scale)*0.875rem);color:var(--text-primary);';
        heSpan.textContent = w.msg || '';      // textContent — XSS safe
        li.appendChild(esSpan);
        li.appendChild(heSpan);
        list.appendChild(li);
      }
      details.appendChild(list);
      panel.appendChild(details);

      // "Estudiar las nuevas" button (SYNC-03)
      const studyBtn = document.createElement('button');
      studyBtn.style.cssText = 'display:block;width:100%;padding:var(--sp-3) var(--sp-4);background:var(--blue-700);color:#fff;border-radius:var(--radius-md);font:700 var(--text-sm) var(--font-body);border:none;cursor:pointer;margin-bottom:var(--sp-3);';
      studyBtn.textContent = '📚 Estudiar las nuevas';
      studyBtn.addEventListener('click', () => {
        overlay.remove();
        // Pre-select "Nuevas" filter in sesion state so session screen shows new words
        if (App.ui._sesionState) {
          App.ui._sesionState.filter = 'nuevas';
          const settings = App.storage ? App.storage.getSettings() : (Store.get('ivrit.v1.settings') || {});
          settings.lastSession = Object.assign({}, App.ui._sesionState, { filter: 'nuevas' });
          Store.set('ivrit.v1.settings', settings);
        }
        location.hash = '#/sesion';
      });
      panel.appendChild(studyBtn);
    }

    // Close button
    const closeBtn = document.createElement('button');
    closeBtn.style.cssText = 'display:block;width:100%;padding:var(--sp-3) var(--sp-4);border:1.5px solid var(--border);border-radius:var(--radius-md);font:600 var(--text-sm) var(--font-body);background:transparent;color:var(--text-secondary);cursor:pointer;';
    closeBtn.textContent = 'Cerrar';
    closeBtn.addEventListener('click', () => overlay.remove());
    panel.appendChild(closeBtn);

    overlay.appendChild(panel);
    // Close on overlay click
    overlay.addEventListener('click', e => { if (e.target === overlay) overlay.remove(); });
    document.body.appendChild(overlay);

    // Inject spin keyframe once
    if (!document.getElementById('sync-keyframes')) {
      const s = document.createElement('style');
      s.id = 'sync-keyframes';
      s.textContent = '@keyframes spin { to { transform: rotate(360deg); } } @keyframes sync-overlay-in { from { opacity:0; } } @keyframes sync-panel-in { from { opacity:0; transform: translateY(12px); } }';
      document.head.appendChild(s);
    }
  };
  ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| App.sync.run() result → DOM | diff.added[].es and diff.added[].msg are remote strings entering the DOM |
| error.message → toast | Error messages enter the DOM via showToast |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-05 | XSS | renderSyncResult — new-word list | mitigate | w.es and w.msg rendered exclusively via `element.textContent = ...` — no innerHTML with remote data anywhere in the result panel. All panel structure built with createElement/appendChild. |
| T-02-06 | XSS | error toast | mitigate | Error messages from { code, message } object. The `msgs` lookup uses hardcoded strings for known codes; only falls back to `err.message` for unknown codes, which are also sanitized by showToast using textContent. |
| T-02-07 | Denial of Service | added-word list | accept | added[] is bounded by vocab size (~735 words max). Rendering 735 li elements is <5ms. No pagination needed. |
| T-02-08 | Spoofing | sync-last-date span | accept | Only date strings (from new Date(updatedAt).toLocaleDateString) are set via textContent. updatedAt is the ISO string from the validated response — already checked by validateRaw in Plan 01. |
</threat_model>
