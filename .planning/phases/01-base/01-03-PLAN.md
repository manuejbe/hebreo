// === ROUTER ===
App.router = {
  current: { route: 'inicio', params: {} },

  _parseHash(hash) {
    // '#/inicio?cat=Verbos' → { route: 'inicio', params: { cat: 'Verbos' } }
    const withoutHash = (hash || '#/inicio').replace(/^#\/?/, '');
    const [routePart, queryPart] = withoutHash.split('?');
    const route = routePart || 'inicio';
    const params = {};
    if (queryPart) {
      queryPart.split('&').forEach(pair => {
        const [k, v] = pair.split('=');
        if (k) params[decodeURIComponent(k)] = decodeURIComponent(v || '');
      });
    }
    return { route, params };
  },

  navigate(hash) {
    const parsed = App.router._parseHash(hash);
    App.router.current = parsed;

    // Update nav active state
    document.querySelectorAll('.nav-item').forEach(el => {
      const r = el.dataset.route;
      // Estudiar nav item activates for sesion and all study modes
      const studyRoutes = ['sesion','tarjetas','opcion-multiple','unir','completar','concordancia','escribir','clasificar','contrarreloj'];
      let active = false;
      if (r === 'sesion') active = studyRoutes.includes(parsed.route);
      else if (r === 'estadisticas') active = parsed.route === 'estadisticas';
      else if (r === 'configuracion') active = parsed.route === 'configuracion';
      else if (r === 'diccionario') active = parsed.route === 'diccionario';
      else if (r === 'inicio') active = parsed.route === 'inicio';
      el.classList.toggle('active', active);
    });

    // Render the screen
    const main = document.getElementById('app-main');

    // Page transition: add leave class to current content
    const existing = main.firstElementChild;
    if (existing) {
      existing.classList.add('page-leave');
      // Wait for animation then replace
      existing.addEventListener('animationend', () => {
        App.router._renderScreen(parsed, main);
      }, { once: true });
      // Fallback if animation doesn't fire (reduced motion)
      setTimeout(() => {
        if (main.firstElementChild === existing) {
          App.router._renderScreen(parsed, main);
        }
      }, 250);
    } else {
      App.router._renderScreen(parsed, main);
    }
  },

  _renderScreen(parsed, main) {
    main.innerHTML = '';
    const screenEl = document.createElement('div');
    screenEl.className = 'screen active page-enter';

    const renderers = {
      inicio:          () => App.ui.renderInicio(screenEl),
      diccionario:     () => App.ui.renderDiccionario(screenEl),
      sesion:          () => App.ui.renderSesion(screenEl),
    };

    const studyStubs = ['tarjetas','opcion-multiple','unir','completar','concordancia','escribir','clasificar','contrarreloj','estadisticas','configuracion'];
    if (renderers[parsed.route]) {
      renderers[parsed.route]();
    } else if (studyStubs.includes(parsed.route)) {
      App.ui.renderStub(screenEl, parsed.route);
    } else {
      App.ui.renderInicio(screenEl); // fallback
    }

    main.appendChild(screenEl);
  },

  init() {
    window.addEventListener('hashchange', () => {
      App.router.navigate(window.location.hash);
    });
    // Render initial route
    const hash = window.location.hash || '#/inicio';
    App.router.navigate(hash);
    // Set default hash if none
    if (!window.location.hash) {
      history.replaceState(null, '', '#/inicio');
    }
  }
};
