```js
// === SEARCH NORMALIZATION ===
function normalizeSearch(str) {
  if (!str) return '';
  return String(str)
    // Strip Hebrew niqqud and cantillation (U+0591–U+05C7)
    .replace(/[֑-ׇ]/g, '')
    // NFD decompose then strip combining diacritical marks (Spanish accents)
    .normalize('NFD').replace(/[̀-ͯ]/g, '')
    .toLowerCase()
    .trim();
}
```

Then add App.ui.renderDiccionario:

```js
// === DICCIONARIO SCREEN ===
App.ui.renderDiccionario = function(el) {
  // Group words by category (preserving original category order from vocab)
  const catOrder = [];
  const catMap = {};
  for (const w of App.state.vocab) {
    if (!catMap[w.cat]) {
      catMap[w.cat] = [];
      catOrder.push(w.cat);
    }
    catMap[w.cat].push(w);
  }

  function renderWordTable(words) {
    if (words.length === 0) return '';
    const kind = words[0].kind;

    // Determine columns based on kind
    let cols, headers;
    if (kind === 'verb') {
      cols = ['es', 'msg'];
      headers = ['Español', '<span lang="he" dir="rtl">פועל</span>'];
    } else if (kind === 'expr') {
      cols = ['es', 'msg'];
      headers = ['Español', '<span lang="he" dir="rtl">ביטוי</span>'];
    } else if (kind === 'country') {
      cols = ['es', 'msg', 'group'];
      headers = ['Español', '<span lang="he" dir="rtl">מדינה</span>', '<span lang="he" dir="rtl">אזור</span>'];
    } else if (kind === 'gendered') {
      cols = ['es', 'msg', 'fsg', 'mpl', 'fpl'];
      headers = ['Español', '<span lang="he" dir="rtl">יחיד</span>', '<span lang="he" dir="rtl">יחידה</span>', '<span lang="he" dir="rtl">רבים</span>', '<span lang="he" dir="rtl">רבות</span>'];
    } else { // noun
      cols = ['es', 'msg', 'mpl'];
      headers = ['Español', '<span lang="he" dir="rtl">יחיד</span>', '<span lang="he" dir="rtl">רבים</span>'];
    }

    const hebrewCols = new Set(['msg', 'fsg', 'mpl', 'fpl', 'group']);

    const rows = words.map(w => {
      const box = (App.state.progress[w.id] && App.state.progress[w.id].box) || 0;
      const cells = cols.map(col => {
        const val = w[col];
        if (col === 'es') {
          return `<td style="padding: var(--sp-2) var(--sp-3); font-size: var(--text-sm); color: var(--text-primary); border-bottom: 1px solid var(--border);">${val || '—'} <span class="box-chip" data-box="${box}" style="margin-left: 4px;">${box}</span></td>`;
        }
        const isHe = hebrewCols.has(col);
        const display = val || '—';
        if (isHe && val) {
          return `<td style="padding: var(--sp-2) var(--sp-3); border-bottom: 1px solid var(--border); text-align: right;">
            <span class="he" lang="he" dir="rtl" style="font-size: calc(var(--he-scale) * 0.875rem);">${display}</span>
            <button onclick="App.ui.showToast('Audio disponible en fases futuras')" aria-label="Pronunciar ${display}"
              style="margin-left: var(--sp-2); background: none; border: none; cursor: pointer; color: var(--text-muted); font-size: 0.875rem; vertical-align: middle;">🔊</button>
          </td>`;
        }
        return `<td style="padding: var(--sp-2) var(--sp-3); border-bottom: 1px solid var(--border); color: var(--text-secondary); font-size: var(--text-sm);">${display}</td>`;
      }).join('');
      return `<tr>${cells}</tr>`;
    }).join('');

    const headerRow = headers.map((h, i) => {
      const isHe = i > 0; // first column is Spanish
      return `<th style="padding: var(--sp-2) var(--sp-3); text-align: ${isHe ? 'right' : 'left'}; font: 600 var(--text-xs) var(--font-body); color: var(--text-muted); border-bottom: 2px solid var(--border); white-space: nowrap;">${h}</th>`;
    }).join('');

    return `<div style="overflow-x: auto; -webkit-overflow-scrolling: touch;">
      <table style="width: 100%; border-collapse: collapse; font-size: var(--text-sm);">
        <thead><tr>${headerRow}</tr></thead>
        <tbody>${rows}</tbody>
      </table>
    </div>`;
  }

  function renderCategoryList(query) {
    const norm = normalizeSearch(query);

    if (!norm) {
      // Show all categories as collapsed accordions (default: closed)
      return catOrder.map(cat => {
        const words = catMap[cat];
        return `
          <details class="dict-cat" style="border: 1px solid var(--border); border-radius: var(--radius-md); margin-bottom: var(--sp-3); overflow: hidden;">
            <summary style="padding: var(--sp-4); cursor: pointer; font: 600 1rem var(--font-display); color: var(--text-primary); display: flex; justify-content: space-between; align-items: center; list-style: none; user-select: none; background: var(--surface);">
              <span>${cat}</span>
              <span style="font: 400 var(--text-sm) var(--font-body); color: var(--text-muted);">${words.length} palabras</span>
            </summary>
            <div style="padding: 0 var(--sp-1) var(--sp-2);">
              ${renderWordTable(words)}
            </div>
          </details>`;
      }).join('');
    }

    // Search mode: flat list grouped by category heading
    let matched = [];
    for (const cat of catOrder) {
      const words = catMap[cat].filter(w =>
        normalizeSearch(w.es).includes(norm) ||
        normalizeSearch(w.msg).includes(norm) ||
        normalizeSearch(w.fsg).includes(norm) ||
        normalizeSearch(w.mpl).includes(norm) ||
        normalizeSearch(w.fpl).includes(norm)
      );
      if (words.length > 0) {
        matched.push({ cat, words });
      }
    }

    if (matched.length === 0) {
      return `<div style="padding: var(--sp-10) var(--sp-4); text-align: center;">
        <p style="color: var(--text-muted); margin-bottom: var(--sp-4);">Sin resultados para "<strong>${query}</strong>"</p>
        <button onclick="document.getElementById('dict-search').value=''; App.ui._dictUpdate('')"
          style="padding: var(--sp-2) var(--sp-5); border: 1.5px solid var(--border); border-radius: var(--radius-pill); background: transparent; color: var(--text-secondary); font: 600 var(--text-sm) var(--font-body); cursor: pointer; min-height: 44px;">
          Limpiar búsqueda
        </button>
      </div>`;
    }

    return matched.map(({ cat, words }) => `
      <div style="margin-bottom: var(--sp-5);">
        <h3 style="font: 600 1rem var(--font-display); color: var(--text-primary); padding: var(--sp-2) 0; border-bottom: 1px solid var(--border); margin-bottom: var(--sp-2);">${cat}</h3>
        ${renderWordTable(words)}
      </div>`).join('');
  }

  el.innerHTML = `
    <div class="screen-diccionario">
      <header style="margin-bottom: var(--sp-5);">
        <h1 style="font: 600 var(--text-4xl) var(--font-display); color: var(--text-primary); margin-bottom: var(--sp-4);">Diccionario</h1>
        <div style="position: relative;">
          <input id="dict-search" type="search" placeholder="Buscar en español o hebreo…"
            oninput="App.ui._dictUpdate(this.value)"
            style="width: 100%; padding: var(--sp-3) var(--sp-4) var(--sp-3) calc(var(--sp-4) + 20px); border: 1.5px solid var(--border); border-radius: var(--radius-pill); background: var(--surface); color: var(--text-primary); font: 400 var(--text-base) var(--font-body); outline: none; transition: border-color var(--dur-fast); min-height: 44px;"
            onfocus="this.style.borderColor='var(--blue-700)'" onblur="this.style.borderColor='var(--border)'"
            aria-label="Buscar palabras">
          <svg style="position: absolute; left: var(--sp-4); top: 50%; transform: translateY(-50%); color: var(--text-muted); pointer-events: none;" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" aria-hidden="true"><circle cx="11" cy="11" r="8"/><line x1="21" y1="21" x2="16.65" y2="16.65"/></svg>
        </div>
      </header>

      <div id="dict-results">
        ${renderCategoryList('')}
      </div>

      <div style="height: var(--sp-8);"></div>
    </div>
  `;

  // Expose update function for oninput handler
  App.ui._dictUpdate = function(query) {
    document.getElementById('dict-results').innerHTML = renderCategoryList(query);
  };
};
```

Note: The `renderCategoryList` and `renderWordTable` functions are defined inside `renderDiccionario` (closure scope), so they can be called from within. The `_dictUpdate` is exposed on App.ui so the inline oninput handler can reach it after re-renders.
