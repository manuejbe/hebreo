function detectKind(fields) {
  // fields = set of header strings in this category's headers array
  if (fields.has('פועל'))  return 'verb';
  if (fields.has('ביטוי')) return 'expr';
  if (fields.has('מדינה')) return 'country';

```js
App.data.normalize = function(rawData) {
  const HEADER_MAP = {
    'תרגום': 'es',
    'יחיד':  'msg',
    'רבים':  'mpl',
    'יחידה': 'fsg',
    'רבות':  'fpl',
    'פועל':  'msg',
    'ביטוי': 'msg',
    'מדינה': 'msg',
    'אזור':  'group',
    'תחום':  'group',
  };

  function slugify(str) {
    return String(str).normalize('NFD').replace(/[̀-ͯ]/g, '')
      .toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  }

  function detectKind(headerSet) {
    if (headerSet.has('פועל'))  return 'verb';
    if (headerSet.has('ביטוי')) return 'expr';
    if (headerSet.has('מדינה')) return 'country';
    if (headerSet.has('יחידה') || headerSet.has('רבות')) return 'gendered';
    return 'noun';
  }

  // Merge categories with duplicate names (SPEC §2.1)
  const mergedCats = [];
  const catIndex = {};
  for (const cat of rawData.categories) {
    if (catIndex[cat.name] !== undefined) {
      // append rows to existing category (headers should match)
      mergedCats[catIndex[cat.name]].rows.push(...cat.rows);
    } else {
      catIndex[cat.name] = mergedCats.length;
      mergedCats.push({ name: cat.name, headers: cat.headers, rows: [...cat.rows] });
    }
  }

  const words = [];
  for (const cat of mergedCats) {
    const catSlug = slugify(cat.name);
    const headerSet = new Set(cat.headers);
    const kind = detectKind(headerSet);
    const esCountInCat = {};

    for (const row of cat.rows) {
      const word = { id: '', cat: cat.name, kind, es: '', msg: null, mpl: null, fsg: null, fpl: null, group: null, extra: {} };

      for (let i = 0; i < cat.headers.length; i++) {
        const header = cat.headers[i];
        const raw = row[i];
        const val = (raw === '—' || raw === undefined || raw === null) ? null : String(raw).trim() || null;
        const field = HEADER_MAP[header];
        if (field) {
          word[field] = val;
        } else {
          word.extra[header] = val;
        }
      }

      // Build stable ID
      const esKey = word.es || '';
      esCountInCat[esKey] = (esCountInCat[esKey] || 0) + 1;
      if (esCountInCat[esKey] === 1) {
        word.id = catSlug + '::' + slugify(esKey);
      } else {
        word.id = catSlug + '::' + slugify(esKey) + '::' + slugify(word.msg || String(esCountInCat[esKey]));
      }

      words.push(word);
    }
  }
  return words;
};
```

Note on duplicate ID handling: the spec says add `"::" + msg` if duplicate. The implementation above uses a count-based approach but that doesn't match the spec exactly for >2 duplicates. Use the spec approach: track seen IDs, and for duplicates add `"::" + msg`.

```js
App.data.load = async function() {
  let raw = null;

  // Try to load vocab.json (fetch, not XHR, works with python3 -m http.server)
  try {
    const resp = await fetch('./vocab.json');
    if (resp.ok) {
      raw = await resp.json();
    }
  } catch (e) {
    // Network/parse error — fall through to cache
  }

  // Check localStorage cache
  const cache = Store.get('ivrit.v1.vocabCache');

  if (raw && cache && cache.updated && raw.updated && cache.updated >= raw.updated) {
    // Cache is newer or same as file — use cache
    raw = cache.raw;
  } else if (raw) {
    // File is newer — update cache with raw JSON
    Store.set('ivrit.v1.vocabCache', { raw: raw, updated: raw.updated || new Date().toISOString() });
  } else if (cache && cache.raw) {
    // File failed, use cache
    raw = cache.raw;
    App.ui.showToast('Usando vocabulario guardado en caché');
  } else {
    // Both failed
    App.ui.showToast('No se pudo cargar el vocabulario');
    App.state.vocab = [];
    return;
  }

  try {
    App.state.vocab = App.data.normalize(raw);
  } catch (e) {
    App.ui.showToast('Error al procesar el vocabulario');
    App.state.vocab = [];
  }
};
```
