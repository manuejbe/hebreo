**3. Theme system:**
```js
App.ui.applyTheme = function(theme) {
  // theme: 'auto' | 'claro' | 'oscuro'
  const html = document.documentElement;
  if (theme === 'oscuro') { html.setAttribute('data-theme', 'dark'); }
  else if (theme === 'claro') { html.setAttribute('data-theme', 'light'); }
  else { html.removeAttribute('data-theme'); } // auto = let CSS media query decide
  App.state.settings.theme = theme;
  Store.set('ivrit.v1.settings', App.state.settings);
};
App.ui.initTheme = function() {
  const settings = Store.get('ivrit.v1.settings') || {};
  const theme = settings.theme || 'auto';
  App.ui.applyTheme(theme);
};
```
Note: For the dark theme CSS, instead of using `@media` inside `[data-theme="dark"]`, structure it as:
- `:root` = light defaults
- `@media (prefers-color-scheme: dark) { :root:not([data-theme="light"]) { ...dark overrides... } }`
- `[data-theme="dark"] { ...dark overrides... }` (explicit dark)
- `[data-theme="light"]` is implicit (no overrides needed; :root defaults are light)

This way: auto = media query applies; oscuro = data-theme=dark applies; claro = data-theme attribute absent + not(light) excludes auto dark.
Actually simpler: use `[data-theme="dark"]` for explicit dark; `@media(prefers-color-scheme:dark){ :root:not([data-theme="light"]) {...} }` for auto dark; no attribute = auto (follows media query).
